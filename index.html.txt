    <script>
        function switchTab(type) {
            const tabs = document.querySelectorAll('.tab-btn');
            tabs.forEach(tab => tab.classList.remove('active'));

            if (type === 'ceilings') {
                tabs[0].classList.add('active');
                document.getElementById('ceilings-calc').style.display = 'grid';
                document.getElementById('doors-calc').style.display = 'none';
            } else {
                tabs[1].classList.add('active');
                document.getElementById('ceilings-calc').style.display = 'none';
                document.getElementById('doors-calc').style.display = 'grid';
            }
        }

        function calculateCeiling() {
            const pricePerMeter = parseInt(document.getElementById('ceiling-type').value);
            const area = parseInt(document.getElementById('ceiling-area').value);
            const lights = parseInt(document.getElementById('ceiling-lights').value);

            document.getElementById('area-val').innerText = area + ' м²';
            document.getElementById('lights-val').innerText = lights + ' шт.';

            const total = (pricePerMeter * area) + (lights * 500); // 500 руб за светильник
            document.getElementById('ceiling-total').innerText = total.toL


ocaleString() + ' ₽';
        }

        function calculateDoors() {
            const pricePerDoor = parseInt(document.getElementById('door-type').value);
            const count = parseInt(document.getElementById('doors-count').value);
            const extra = parseInt(document.getElementById('door-extra').value);

            document.getElementById('doors-count-val').innerText = count + ' шт.';

            const total = (pricePerDoor + extra) * count;
            document.getElementById('doors-total').innerText = total.toLocaleString() + ' ₽';
        }

        function openOrderModal(service) {
            alert('Спасибо за выбор! Для фиксации стоимости услуги "' + service + '" укажите имя и телефон в форме внизу страницы.');
            document.querySelector('.contact-section').scrollIntoView({ behavior: 'smooth' });
        }

        function submitForm(event) {
            event.preventDefault();
            alert('Спасибо за заявку! Наш замерщик свяжется с вами в блищайшее время.');
        }

        // Инициализация стартовых расчетов
        calculateCeiling();
        calculateDoors();
    </script>
